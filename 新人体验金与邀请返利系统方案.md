**架构**: Monorepo (server/web/mobile)

## 第一部分：新人首单体验金（10元）

### 1. 业务规则

- 注册成功后自动发放10元体验金
- 有效期30天，过期自动回收（本金不返还）
- 可用于认购，但 **不可提现**
- 体验金产生的收益可提现
- 体验金优先使用（认购时先用真实余额，再用体验金）

```typescript
import { Injectable, Logger, BadRequestException, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, DataSource, LessThanOrEqual } from 'typeorm';
import { Cron } from '@nestjs/schedule';
import { TrialBonus, TrialBonusStatus } from '../../database/entities/trial-bonus.entity';
import { AccountBalance } from '../../database/entities/account-balance.entity';
import { AccountTransaction, TransactionType } from '../../database/entities/account-transaction.entity';

@Injectable()
export class TrialBonusService {
  private logger = new Logger(TrialBonusService.name);

  constructor(
    @InjectRepository(TrialBonus) private trialBonusRepository: Repository<TrialBonus>,
    @InjectRepository(AccountBalance) private accountBalanceRepository: Repository<AccountBalance>,
    @InjectRepository(AccountTransaction) private accountTransactionRepository: Repository<AccountTransaction>,
    private dataSource: DataSource,
  ) {}

  // 为新用户发放体验金（注册后调用）
  async grantTrialBonus(userId: string, amount: number = 10): Promise<TrialBonus> {
    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    await queryRunner.startTransaction();

    try {
      // 检查是否已发放
      const existing = await queryRunner.manager.findOne(TrialBonus, {
        where: { userId, status: TrialBonusStatus.ACTIVE },
      });
      if (existing) throw new BadRequestException('已获得体验金');

      // 获取账户余额
      let balance = await queryRunner.manager.findOne(AccountBalance, {
        where: { userId },
        lock: { mode: 'pessimistic_write' },
      });

      if (!balance) throw new NotFoundException('账户不存在');

      // 计算过期时间
      const grantedAt = new Date();
      const expiresAt = new Date(grantedAt);
      expiresAt.setDate(expiresAt.getDate() + 30);

      // 创建体验金记录
      const trialBonus = queryRunner.manager.create(TrialBonus, {
        userId,
        initialAmount: amount,
        usedAmount: 0,
        remainingAmount: amount,
        status: TrialBonusStatus.ACTIVE,
        grantedAt,
        expiresAt,
      });
      const saved = await queryRunner.manager.save(trialBonus);

      // 更新余额
      const before = Number(balance.trialBalance);
      balance.trialBalance = Number((before + amount).toFixed(2));
      balance.trialExpiresAt = expiresAt;
      balance.trialExpired = false;
      await queryRunner.manager.save(balance);

      // 记录交易
      const tx = queryRunner.manager.create(AccountTransaction, {
        userId,
        type: TransactionType.TRIAL_BONUS_GRANT,
        amount,
        balanceBefore: before,
        balanceAfter: balance.trialBalance,
        description: `新用户注册奖励：体验金¥${amount}`,
      });
      await queryRunner.manager.save(tx);

      await queryRunner.commitTransaction();
      return saved;
    } catch (e) {
      await queryRunner.rollbackTransaction();
      throw e;
    } finally {
      await queryRunner.release();
    }
  }

  // 定时任务：每天凌晨2点处理过期
  @Cron('0 2 * * *')
  async handleExpiration(): Promise<void> {
    this.logger.log('处理体验金过期...');
    const now = new Date();
    const queryRunner = this.dataSource.createQueryRunner();

    try {
      const expired = await queryRunner.manager.find(TrialBonus, {
        where: { status: TrialBonusStatus.ACTIVE, expiresAt: LessThanOrEqual(now) },
      });

      for (const bonus of expired) {
        await queryRunner.startTransaction();
        try {
          bonus.status = TrialBonusStatus.EXPIRED;
          bonus.expiredAt = now;
          await queryRunner.manager.save(bonus);

          const balance = await queryRunner.manager.findOne(AccountBalance, {
            where: { userId: bonus.userId },
            lock: { mode: 'pessimistic_write' },
          });

          if (balance) {
            const before = Number(balance.trialBalance);
            balance.trialBalance = 0;
            balance.trialExpired = true;
            await queryRunner.manager.save(balance);

            const tx = queryRunner.manager.create(AccountTransaction, {
              userId: bonus.userId,
              type: TransactionType.TRIAL_BONUS_EXPIRE,
              amount: -before,
              balanceBefore: before,
              balanceAfter: 0,
              description: `体验金过期回收：¥${before.toFixed(2)}`,
            });
            await queryRunner.manager.save(tx);
          }
          await queryRunner.commitTransaction();
        } catch (e) {
          await queryRunner.rollbackTransaction();
          this.logger.error(`处理用户 ${bonus.userId} 过期失败`, e);
        }
      }
    } finally {
      await queryRunner.release();
    }
  }
}
```

## 第二部分：邀请返利系统

### 1. 业务规则

- 每用户自动生成6位唯一邀请码（首次登录时）
- 新用户注册时可填入邀请码
- 邀请成功后：邀请人+5元，被邀请人+3元（首次认购时发放）
- 防刷：同IP 24h限5个账户，同手机1个账户，邀请码100个账户上限
- 状态：PENDING(待认购) → COMPLETED(已完成) → CANCELLED(已失效)

#### 3.1 InvitationService

```typescript
import { Injectable, BadRequestException, NotFoundException, ConflictException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, DataSource } from 'typeorm';
import { InvitationCode } from '../../database/entities/invitation-code.entity';
import { InvitationRecord, InvitationStatus } from '../../database/entities/invitation-record.entity';
import { AccountBalance } from '../../database/entities/account-balance.entity';
import { AccountTransaction, TransactionType } from '../../database/entities/account-transaction.entity';

@Injectable()
export class InvitationService {
  constructor(
    @InjectRepository(InvitationCode) private codeRepository: Repository<InvitationCode>,
    @InjectRepository(InvitationRecord) private recordRepository: Repository<InvitationRecord>,
    @InjectRepository(AccountBalance) private balanceRepository: Repository<AccountBalance>,
    @InjectRepository(AccountTransaction) private txRepository: Repository<AccountTransaction>,
    private dataSource: DataSource,
  ) {}

  // 为新用户生成邀请码
  async generateInvitationCode(userId: string): Promise<InvitationCode> {
    const existing = await this.codeRepository.findOne({ where: { userId } });
    if (existing) return existing;

    let code: string, isUnique = false, retries = 0;
    while (!isUnique && retries < 10) {
      code = this.generateCode();
      const found = await this.codeRepository.findOne({ where: { code } });
      if (!found) isUnique = true;
      retries++;
    }

    if (!isUnique) throw new BadRequestException('生成邀请码失败');

    return this.codeRepository.save({ userId, code, usedCount: 0 });
  }

  private generateCode(): string {
    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
    let code = '';
    for (let i = 0; i < 6; i++) code += chars[Math.floor(Math.random() * 36)];
    return code;
  }

  // 绑定邀请关系（注册时）
  async bindInvitationCode(userId: string, invitationCode: string): Promise<InvitationRecord> {
    const code = await this.codeRepository.findOne({ where: { code: invitationCode.toUpperCase() } });
    if (!code) throw new NotFoundException('邀请码不存在');
    if (code.userId === userId) throw new BadRequestException('不能使用自己的邀请码');

    const existing = await this.recordRepository.findOne({ where: { invitedUserId: userId } });
    if (existing) throw new ConflictException('已被邀请过');

    if (code.usedCount >= 100) throw new BadRequestException('邀请码已达上限');

    const record = this.recordRepository.create({
      inviterUserId: code.userId,
      invitedUserId: userId,
      codeId: code.id,
      status: InvitationStatus.PENDING,
    });
    const saved = await this.recordRepository.save(record);

    code.usedCount += 1;
    await this.codeRepository.save(code);

    return saved;
  }

  // 发放奖励（首次认购时触发）
  async grantBonus(userId: string, subscriptionOrderId: string): Promise<void> {
    const record = await this.recordRepository.findOne({
      where: { invitedUserId: userId, status: InvitationStatus.PENDING },
    });

    if (!record) return; // 无有效邀请关系

    const queryRunner = this.dataSource.createQueryRunner();
    await queryRunner.connect();
    await queryRunner.startTransaction();

    try {
      // 邀请人账户
      const inviterBal = await queryRunner.manager.findOne(AccountBalance, {
        where: { userId: record.inviterUserId },
        lock: { mode: 'pessimistic_write' },
      });

      // 被邀请人账户
      const invitedBal = await queryRunner.manager.findOne(AccountBalance, {
        where: { userId: record.invitedUserId },
        lock: { mode: 'pessimistic_write' },
      });

      if (!inviterBal || !invitedBal) throw new NotFoundException('账户不存在');

      // 发放邀请人奖励
      const inviterAmount = Number(record.inviterBonus);
      const inviterBefore = Number(inviterBal.availableBalance);
      inviterBal.availableBalance = Number((inviterBefore + inviterAmount).toFixed(2));
      await queryRunner.manager.save(inviterBal);

      const inviterTx = queryRunner.manager.create(AccountTransaction, {
        userId: record.inviterUserId,
        type: TransactionType.INVITATION_INVITER_BONUS,
        amount: inviterAmount,
        balanceBefore: inviterBefore,
        balanceAfter: inviterBal.availableBalance,
        description: `邀请奖励：用户 ${record.invitedUserId} 完成首次认购，获得¥${inviterAmount}`,
      });
      await queryRunner.manager.save(inviterTx);

      // 发放被邀请人奖励
      const invitedAmount = Number(record.invitedBonus);
      const invitedBefore = Number(invitedBal.availableBalance);
      invitedBal.availableBalance = Number((invitedBefore + invitedAmount).toFixed(2));
      await queryRunner.manager.save(invitedBal);

      const invitedTx = queryRunner.manager.create(AccountTransaction, {
        userId: record.invitedUserId,
        type: TransactionType.INVITATION_INVITED_BONUS,
        amount: invitedAmount,
        balanceBefore: invitedBefore,
        balanceAfter: invitedBal.availableBalance,
        description: `邀请奖励：首次认购完成，获得¥${invitedAmount}`,
      });
      await queryRunner.manager.save(invitedTx);

      // 更新邀请记录
      record.status = InvitationStatus.COMPLETED;
      record.firstSubscriptionOrderId = subscriptionOrderId;
      record.bonusGrantedAt = new Date();
      await queryRunner.manager.save(record);

      await queryRunner.commitTransaction();
    } catch (e) {
      await queryRunner.rollbackTransaction();
      throw e;
    } finally {
      await queryRunner.release();
    }
  }
}
```

### 关键风险和缓解方案

| 风险 | 影响 | 缓解方案 |
|------|------|---------|
| 体验金过期定时任务延迟 | 用户可能在过期后仍能用体验金 | Cron任务配合实时检查 |
| 邀请奖励重复发放 | 双重计费 | 使用status标志和幂等检查 |
| 余额精度问题 | 小数位错误导致不平衡 | 使用DECIMAL(12,2)类型，所有操作保留两位小数 |
| 高并发下的竞态条件 | 多个请求同时使用邀请码 | 使用数据库锁和唯一约束 |
